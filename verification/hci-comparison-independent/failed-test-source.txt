/** Independent literal binary fixtures and positional oracle, not builder helpers. */
import {test} from 'node:test';
import assert from 'node:assert/strict';
import {compareHciLegacyReportBytes as compare, serializeHciReportComparison} from '../packages/core/hci-report-comparison.ts';
import {hash,doc,report} from './hci-comparison-review-fixture.ts';
function oracle(a:number[],b:number[],ao=54,bo=54){return Array.from({length:Math.max(a.length,b.length)},(_,i)=>({dataByteOffset:i,leftHex:i<a.length?a[i]!.toString(16).padStart(2,'0'):null,rightHex:i<b.length?b[i]!.toString(16).padStart(2,'0'):null,leftDerivativeOffset:i<a.length?ao+i:null,rightDerivativeOffset:i<b.length?bo+i:null,relation:i>=a.length?'right-only':i>=b.length?'left-only':a[i]===b[i]?'equal':'different'}));}
function frozen(x:any){if(x&&typeof x==='object'){assert.ok(Object.isFrozen(x));Object.values(x).forEach(frozen);}}
test('independent exhaustive 256 byte values at all 31 positions and null boundaries',async()=>{
 for(let value=0;value<256;value++){
  const a=Array(31).fill(value),b=Array.from({length:31},(_,i)=>(value+i)%256);
  const x=await compare(await doc([report(a)]),0,await doc([report(b)]),0,hash);
  assert.deepEqual(x.rows,oracle(a,b));assert.equal(x.dataBytesEqual,false);frozen(x);
 }
 for(const [a,b] of [[[],[]],[[],[0]],[[0],[]],[[1,2,3],[1,0,2,3]],[[1,0,2,3],[1,2,3]]] as [number[],number[]][]){
  const x=await compare(await doc([report(a)]),0,await doc([report(b)]),0,hash);assert.deepEqual(x.rows,oracle(a,b));assert.equal(x.dataBytesEqual,a.length===0&&b.length===0);
 }
});
test('explicit report 24 and nonfirst offsets; no unselected bytes or documents leak',async()=>{
 const many=await doc([...Array.from({length:24},()=>report([])),report([2,1,6])]);
 const other=await doc([report([7,0x16,0xde,0xad,0xbe,0xef,0x99,0x88]),report([2,1,6],4,9)]);
 const x=await compare(many,24,other,1,hash);assert.deepEqual(x.rows,oracle([2,1,6],[2,1,6],294,72));
 assert.equal(x.left.reportDerivativeOffset,285);assert.equal(x.right.reportDerivativeOffset,63);assert.equal(x.left.rssiRaw,192);assert.equal(x.left.rssiDbm,-64);assert.equal(x.right.kind,'scan-response');assert.equal(x.left.derivative.byteCount,298);assert.match(x.left.derivative.sha256,/^[0-9a-f]{64}$/);assert.equal(x.left.adInterpretation.status,'complete');assert.equal(x.left.reportCount,25);assert.equal(x.left.reportIndex,24);assert.equal(x.right.reportCount,2);assert.equal(x.dataBytesEqual,true);
 assert.deepEqual(x.metadataEquality,{eventType:false,kind:false,addressType:true,addressWireHex:false});
 const out=serializeHciReportComparison(x);for(const marker of ['deadbeef9988','recordHex','fileHeaderHex','"document"'])assert.ok(!out.includes(marker));
 assert.match(x.warnings.join(' '),/independent observations/);assert.match(x.warnings.join(' '),/field meanings/);
});
test('identical pairs are hashed independently sequentially; mutation cannot alter retained data',async()=>{
 const d=await doc([report([2,1,6])]);let active=0,calls=0;
 const x=await compare(d,0,d,0,async b=>{assert.equal(active++,0);calls++;const h=await hash(b);b.fill(0);await Promise.resolve();active--;return h;});
 assert.equal(calls,2);assert.deepEqual(x.rows,oracle([2,1,6],[2,1,6]));
 const v=JSON.parse(d);v.sourceReference={sha256:'b'.repeat(64),byteCount:1000,recordIndex:7,recordOffset:500};v.context.label='changed declaration';
 const y=await compare(d,0,JSON.stringify(v),0,hash);assert.equal(y.dataBytesEqual,true);assert.equal(y.right.sourceReference.verification,'UNVERIFIED');assert.equal(y.right.context.label,'changed declaration');
});
test('primitive bounds and ambiguous JSON reject before any hashing on either side',async()=>{
 const d=await doc([report([])]);let calls=0;const spy=async(b:Uint8Array)=>{calls++;return hash(b);};
 for(const bad of [new String(d),null,{},d+' '.repeat(4097),d.replace('"schemaVersion":1','"schemaVersion":1,"\\u0073chemaVersion":1')])for(const side of [0,1])await assert.rejects(compare(side===0?bad as any:d,0,side===1?bad as any:d,0,spy));
 for(const bad of [-0,-1,25,NaN,Infinity,0.5,'0',new Number(0),null])for(const side of [0,1])await assert.rejects(compare(d,side===0?bad as any:0,d,side===1?bad as any:0,spy));
 assert.equal(calls,0);const padded=d+' '.repeat(4096-d.length);assert.equal((await compare(padded,0,padded,0,spy)).rows.length,0);assert.equal(calls,2);
});
test('whole event late invalid unselected report refuses despite matching recomputed digest',async()=>{
 const good=await doc([report([0])]),bad=await doc([report([0]),report([],5)]);
 for(const [a,b]of [[good,bad],[bad,good]])await assert.rejects(compare(a!,0,b!,0,hash));
 for(const cb of [async()=>{throw Error('PRIVATE HASH ERROR');},async()=> 'A'.repeat(64),async()=> '0'.repeat(64)])await assert.rejects(compare(good,0,good,0,cb));
 let calls=0;await assert.rejects(compare(good,1,good,0,async b=>{calls++;return hash(b);}));assert.equal(calls,2);
});
test('serializer counts final newline UTF8 bytes and refuses oversized output',async()=>{
 const d=await doc([report([])]),x=await compare(d,0,d,0,hash);const empty={...x,warnings:['']};const size=Buffer.byteLength(serializeHciReportComparison(empty)+'\n');
 const exact={...x,warnings:['a'.repeat(65536-size)]};assert.equal(Buffer.byteLength(serializeHciReportComparison(exact)+'\n'),65536);
 assert.throws(()=>serializeHciReportComparison({...x,warnings:['a'.repeat(65537-size)]}));assert.throws(()=>serializeHciReportComparison({...x,warnings:['漢'.repeat(30000)]}));
});
