import test from 'node:test';
import assert from 'node:assert/strict';
import {flipperWorkbenchReducer as r, initialFlipperWorkbench as initial, flipperVisibleFrames, flipperVisibleFields, FLIPPER_RESPONSE_FIXTURE as fixture, FLIPPER_TEXT_LIMIT} from '../../src/flipper-workbench.ts';
const inspect=()=>r(r(initial,{type:'fixture'}),{type:'inspect'});
test('report returned by visible frames cannot mutate retained report',()=>{
 const s=inspect(); const exposed=flipperVisibleFrames(s);
 try {exposed[0]!.status.effective=99;} catch {}
 assert.equal(s.report!.frames[0]!.status.effective,0);
});
test('original bytes exposed through state cannot alter report evidence',()=>{
 const s=inspect(); try {s.report!.originalBytes.fill(0);} catch {}
 assert.equal(s.report!.originalBytes[0],13);
});
test('page transitions cannot mutate reports retained by earlier states',()=>{
 const s=inspect(); const next=r(s,{type:'frame-page',page:0});
 try {next.report!.frames[0]!.content.major!.effective=999;} catch {}
 assert.equal(s.report!.frames[0]!.content.major!.effective,1);
});
test('field helper metadata is detached',()=>{
 const s=inspect(); const fields=flipperVisibleFields(s);fields[0]!.offset=999;
 assert.equal(s.report!.frames[0]!.fields[0]!.offset,1);
});
test('accessor edit cannot validate one draft and retain another',()=>{
 let reads=0;
 const action={type:'edit' as const,get draft(){return ++reads===1?fixture:'x'.repeat(FLIPPER_TEXT_LIMIT+1);}};
 const s=r(inspect(),action);
 assert.ok(s.draft.length<=FLIPPER_TEXT_LIMIT);
});
test('invalid edit clears stale report and page indexes',()=>{
 for(const draft of ['f','0\t0','00\u0085','00\u2028','00\u200b','00\u0000','0x00','00'.repeat(65537)]){
 const s=r(inspect(),{type:'edit',draft});assert.equal(s.report,null);assert.equal(s.draft,fixture);assert.equal(s.framePage,0);assert.equal(s.selectedFrame,0);assert.equal(s.fieldPage,0);
 }
});
test('provenance action snapshots the declaration before validation',()=>{
 let reads=0;
 const action={type:'provenance' as const,get provenance(){return (++reads===1?'synthetic':'verified-device') as 'synthetic';}};
 const s=r(inspect(),action);
 assert.ok(s.provenance===null||s.provenance==='synthetic'||s.provenance==='imported');
});
test('invalid pure navigation inputs retain current bounded state',()=>{
 const s=inspect();
 for(const value of [-0.5,-1,NaN,Infinity,Number.MAX_SAFE_INTEGER,'0',null,{},undefined]){
 for(const type of ['frame-page','field-page'] as const) assert.equal(r(s,{type,page:value as number}),s);
 assert.equal(r(s,{type:'select-frame',index:value as number}),s);
 }
});
